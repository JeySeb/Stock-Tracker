func setupTestDatabase(t *testing.T) *sql.DB {
	// Setup test database connection
	db, err := database.NewConnection(database.Config{
		Host:     "localhost",
		Port:     5432,
		User:     "test_user",
		Password: "test_password",
		DBName:   "test_db",
		SSLMode:  "disable",
	})
	require.NoError(t, err)

func setupTestServer(t *testing.T, db *sql.DB) (*httptest.Server, *handlers.RecommendationHandler) {
	// Setup dependencies
	testLogger := logger.NewLogger("test")
	stockRepo := database.NewStockRepository(db, testLogger)
	memCache := cache.NewInMemoryCache()

	// Setup mock external API servers
	yahooServer := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		json.NewEncoder(w).Encode(map[string]interface{}{
			"chart": map[string]interface{}{
				"result": []map[string]interface{}{
					{
						"meta": map[string]interface{}{
							"regularMarketPrice":  170.0,
							"previousClose":       168.0,
							"regularMarketVolume": 1000000,
							"marketCap":           2800000000000,
						},
					},
				},
			},
		})
	}))
	defer yahooServer.Close()

	alphaServer := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		json.NewEncoder(w).Encode(map[string]interface{}{
			"Global Quote": map[string]interface{}{
				"05. price": "171.00",
			},
		})
	}))
	defer alphaServer.Close()

	// Setup clients
	yahooClient := external.NewYahooFinanceClient(yahooServer.URL, testLogger)
	alphaClient := external.NewAlphaVantageClient("test_key", testLogger)

	// Setup components
	calculator := recommendationValidation.NewBasicScoringCalculator(stockRepo, testLogger)
	enricher := recommendationValidation.NewExternalDataEnricher(yahooClient, alphaClient, memCache, testLogger)
	useCase := recommendationUsecases.NewTieredRecommendationUseCase(stockRepo, calculator, enricher, memCache, testLogger)
	handler := handlers.NewRecommendationHandler(useCase, testLogger)

	// Setup router with middleware
	router := chi.NewRouter()
	router.Use(middleware.WithUserTier)
	router.Use(middleware.WithRateLimit)

	// Register routes
	router.Get("/api/v1/recommendations", handler.GetRecommendations)
	router.Get("/api/v1/recommendations/{ticker}", handler.GetRecommendationByTicker)
	router.Get("/api/v1/recommendations/preview/{ticker}", handler.GetRecommendationPreview)

	// Create test server
	server := httptest.NewServer(router)

	return server, handler
}

func seedTestData(t *testing.T, db *sql.DB) {
	// Insert test stock events
	events := []stockModel.Stock{
		{
			Ticker:     "AAPL",
			Company:    "Apple Inc.",
			Brokerage:  "High Activity Broker",
			Action:     "target raised by",
			RatingFrom: "Buy",
			RatingTo:   "Strong Buy",
			TargetFrom: 170.0,
			TargetTo:   190.0,
			EventTime:  time.Now().Add(-24 * time.Hour),
		},
		{
			Ticker:     "AAPL",
			Company:    "Apple Inc.",
			Brokerage:  "Another Broker",
			Action:     "upgraded to",
			RatingFrom: "Hold",
			RatingTo:   "Buy",
			TargetFrom: 165.0,
			TargetTo:   180.0,
			EventTime:  time.Now().Add(-48 * time.Hour),
		},
		{
			Ticker:     "GOOGL",
			Company:    "Alphabet Inc.",
			Brokerage:  "High Activity Broker",
			Action:     "target lowered by",
			RatingFrom: "Buy",
			RatingTo:   "Hold",
			TargetFrom: 150.0,
			TargetTo:   140.0,
			EventTime:  time.Now().Add(-24 * time.Hour),
		},
	}

	for _, event := range events {
		_, err := db.Exec(`
			INSERT INTO stocks (ticker, company, brokerage, action, rating_from, rating_to, target_from, target_to, event_time)
			VALUES ($1, $2, $3, $4, $5, $6, $7, $8, $9)
		`,
			event.Ticker, event.Company, event.Brokerage, event.Action,
			event.RatingFrom, event.RatingTo, event.TargetFrom, event.TargetTo, event.EventTime,
		)
		require.NoError(t, err)
	}
}

func TestRecommendationEngine_EndToEnd_GuestUser(t *testing.T) {
	// Setup
	db := setupTestDatabase(t)
	defer db.Close()

	seedTestData(t, db)

	server, _ := setupTestServer(t, db)
	defer server.Close()

	// Test getting recommendations as guest user
	resp, err := http.Get(server.URL + "/api/v1/recommendations")
	require.NoError(t, err)
	defer resp.Body.Close()

	assert.Equal(t, http.StatusOK, resp.StatusCode)

	var response recommendationUsecases.RecommendationResponse
	err = json.NewDecoder(resp.Body).Decode(&response)
	require.NoError(t, err)

	// Verify response
	assert.NotEmpty(t, response.Data)
	assert.Equal(t, enums.TIER_GUEST, response.Meta.UserTier)
	assert.Contains(t, response.Meta.Features, "basic_recommendations")
	assert.NotContains(t, response.Meta.Features, "real_time_data")

	// Verify recommendation data
	firstRec := response.Data[0]
	assert.NotEmpty(t, firstRec.Ticker)
	assert.NotEmpty(t, firstRec.CompanyName)
	assert.NotNil(t, firstRec.BasicScore)
	assert.Equal(t, enums.RECOMMENDATION_TIER_BASIC, firstRec.Tier)
	assert.Nil(t, firstRec.ExternalData) // Guest users don't get external data
}

func TestRecommendationEngine_EndToEnd_BasicUser(t *testing.T) {

	firstRec := response.Data[0]
	assert.Equal(t, enums.RECOMMENDATION_TIER_ENRICHED, firstRec.Tier)
	assert.NotNil(t, firstRec.ExternalData)
	assert.NotZero(t, firstRec.ExternalData.CurrentPrice)
}

			assert.Equal(t, enums.RECOMMENDATION_TYPE_STRONG_BUY, rec.RecommendationType)
		}
	}
}

func TestRecommendationEngine_EndToEnd_Caching(t *testing.T) {
	// Setup
	db := setupTestDatabase(t)
	defer db.Close()

	seedTestData(t, db)

	server, _ := setupTestServer(t, db)
	defer server.Close()

	// First request
	resp1, err := http.Get(server.URL + "/api/v1/recommendations")
	require.NoError(t, err)
	defer resp1.Body.Close()

	var response1 recommendationUsecases.RecommendationResponse
	err = json.NewDecoder(resp1.Body).Decode(&response1)
	require.NoError(t, err)
	assert.False(t, response1.Meta.CacheHit)

	// Second request (should be cached)
	resp2, err := http.Get(server.URL + "/api/v1/recommendations")
	require.NoError(t, err)
	defer resp2.Body.Close()

	var response2 recommendationUsecases.RecommendationResponse
	err = json.NewDecoder(resp2.Body).Decode(&response2)
	require.NoError(t, err)
	assert.True(t, response2.Meta.CacheHit)

	// Verify responses are identical
	assert.Equal(t, len(response1.Data), len(response2.Data))
	assert.Equal(t, response1.Data[0].BasicScore, response2.Data[0].BasicScore)
}

	for i := 0; i < 10; i++ {

			assert.Equal(t, http.StatusOK, resp.StatusCode)

}

func TestRecommendationEngine_EndToEnd_ErrorHandling(t *testing.T) {
	// Setup
	db := setupTestDatabase(t)
	defer db.Close()

	seedTestData(t, db)

	server, _ := setupTestServer(t, db)
	defer server.Close()

	tests := []struct {
		name           string
		endpoint       string
		expectedStatus int
		errorContains  string
	}{
		{
			name:           "Invalid ticker",
			endpoint:       "/api/v1/recommendations/INVALID",
			expectedStatus: http.StatusNotFound,
			errorContains:  "not found",
		},
		{
			name:           "Invalid filter value",
			endpoint:       "/api/v1/recommendations?min_score=invalid",
			expectedStatus: http.StatusBadRequest,
			errorContains:  "invalid parameter",
		},
		{
			name:           "Missing required parameter",
			endpoint:       "/api/v1/recommendations/preview/",
			expectedStatus: http.StatusNotFound,
			errorContains:  "not found",
		},
	}

	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			resp, err := http.Get(server.URL + tt.endpoint)
			require.NoError(t, err)
			defer resp.Body.Close()

			assert.Equal(t, tt.expectedStatus, resp.StatusCode)

			var errorResponse map[string]string
			err = json.NewDecoder(resp.Body).Decode(&errorResponse)
			require.NoError(t, err)
			assert.Contains(t, errorResponse["error"], tt.errorContains)
		})
	}
}

func TestRecommendationEngine_EndToEnd_Performance(t *testing.T) {
	// Setup
	db := setupTestDatabase(t)
	defer db.Close()

	// Seed larger dataset
	for i := 0; i < 100; i++ {
		event := stockModel.Stock{
			Ticker:    "TEST" + string(rune(i)),
			Company:   "Test Company " + string(rune(i)),
			Brokerage: "Test Broker",
			Action:    "target raised by",
			EventTime: time.Now().Add(-time.Duration(i) * time.Hour),
		}
		_, err := db.Exec(`
			INSERT INTO stocks (ticker, company, brokerage, action, event_time)
			VALUES ($1, $2, $3, $4, $5)
		`, event.Ticker, event.Company, event.Brokerage, event.Action, event.EventTime)
		require.NoError(t, err)
	}

	server, _ := setupTestServer(t, db)
	defer server.Close()

	// Test response time
	start := time.Now()
	resp, err := http.Get(server.URL + "/api/v1/recommendations?limit=50")
	require.NoError(t, err)
	defer resp.Body.Close()

	duration := time.Since(start)

	// Verify performance
	assert.Less(t, duration, 2*time.Second, "Response time should be under 2 seconds")
	assert.Equal(t, http.StatusOK, resp.StatusCode)

	var response recommendationUsecases.RecommendationResponse
	err = json.NewDecoder(resp.Body).Decode(&response)
	require.NoError(t, err)
	assert.NotEmpty(t, response.Data)
}
