// Test GetMarketDataAnalysis
func TestMarketDataHandler_GetMarketDataAnalysis_Success(t *testing.T) {
	// Arrange
	mockUseCase := &SimpleMockMarketDataUseCase{}
	mockLogger := &mocks.MockLogger{}

	// Create a real use case instance for the handler
	realUseCase := &marketDataUsecases.MarketDataAnalysisUseCase{}
	handler := handlers.NewMarketDataHandler(*realUseCase, mockLogger)

	analysis := createMockMarketDataAnalysis("AAPL")
	response := createMockMarketDataResponse(analysis, "Market data analysis retrieved successfully")

	mockUseCase.On("GetMarketDataAnalysis", mock.Anything, "AAPL").Return(analysis, nil)
	mockUseCase.On("CreateMarketDataResponse", analysis, mock.Anything, "Market data analysis retrieved successfully").Return(response)

	req := httptest.NewRequest("GET", "/market-data/analysis/AAPL", nil)
	chiCtx := chi.NewRouteContext()
	chiCtx.URLParams.Add("ticker", "AAPL")
	req = req.WithContext(context.WithValue(req.Context(), chi.RouteCtxKey, chiCtx))
	w := httptest.NewRecorder()

	// Act
	handler.GetMarketDataAnalysis(w, req)

	// Assert
	assert.Equal(t, http.StatusOK, w.Code)

	var responseBody marketDataModel.MarketDataResponse
	err := json.NewDecoder(w.Body).Decode(&responseBody)
	require.NoError(t, err)

	assert.Equal(t, "Market data analysis retrieved successfully", responseBody.Message)
	assert.NotNil(t, responseBody.Data)

	mockUseCase.AssertExpectations(t)
}
